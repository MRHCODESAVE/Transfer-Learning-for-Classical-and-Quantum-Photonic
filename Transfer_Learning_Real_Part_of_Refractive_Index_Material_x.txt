% Part 1: Knowledge Transferring
load net_real.mat;
net_real_TL=net_real;
% Part 2: Choose the activation function, and learning rate for 'traingd'
net_real_TL.trainFcn = 'traingd';
net_real_TL.trainParam.lr=0.01;
% Part 3: Choose a performance function for the network, while here is mean squared error
net_real_TL.performFcn = 'mse'; 
% Part 4: Dataset portions for training, validation, testing datasets, the numbers of epochs
net_real_TL.divideParam.trainRatio = 0.7;
net_real_TL.divideParam.valRatio = 0.15;
net_real_TL.divideParam.testRatio = 0.15;
net_real_TL.trainParam.epochs=100000;
% Part 5: Training the neural network (i.e., net_real_TL)
[net_real_TL,training_info] = train(net_real_TL,input_new',N_real_new_noise');
% Part 6: Regression plot
figure
plotregression(N_real_new_noise',net_real_TL(input_new'))
h = findobj('Marker', 'o');
set(h,'Marker','.', 'MarkerSize',6)
grid on
% Part 7: Error plot
figure
plot(input_new',N_real_new_noise'-net_real_TL(input_new'),'b-')
xlabel('Energy (eV)')
ylabel('Errors for the Real Part')
grid on
